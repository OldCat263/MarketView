## #1 `us.py` — 美股代码列表磁盘缓存

**根因**：`ak.stock_us_spot_em()` 拉全量美股行情只为取代码，60-120s。重启缓存丢失。

**改法**：

```python
# us.py 文件头部增加
import os, json

_US_CODES_FILE = os.path.join(os.path.dirname(os.path.abspath(__file__)), '..', '.cache', 'us_codes.json')

# _load_us_codes() 函数改为：
def _load_us_codes():
    global _us_codes_cache, _us_codes_ts
    now = time.time()
    if _us_codes_cache and now - _us_codes_ts < 86400:
        return _us_codes_cache
    # 尝试磁盘缓存
    if _us_codes_cache is None and os.path.exists(_US_CODES_FILE):
        try:
            with open(_US_CODES_FILE, 'r', encoding='utf-8') as f:
                _us_codes_cache = json.load(f)
            _us_codes_ts = now
            return _us_codes_cache
        except Exception:
            pass
    # 回退到 AkShare
    try:
        df = ak.stock_us_spot_em()
        codes = [str(r['代码']) for _, r in df.iterrows()]
    except Exception:
        df = ak.stock_us_spot()
        codes = [str(r.get('symbol','')) for _, r in df.iterrows() if r.get('symbol')]
    _us_codes_cache = codes
    _us_codes_ts = now
    # 持久化
    try:
        os.makedirs(os.path.dirname(_US_CODES_FILE), exist_ok=True)
        with open(_US_CODES_FILE, 'w', encoding='utf-8') as f:
            json.dump(codes, f)
    except Exception:
        pass
    return codes
```

**验收**：首次拉取后检查 `backend/.cache/us_codes.json` 存在，重启服务后美股秒级加载。

## #2 `hk.py` — 港股腾讯并行路径

**根因**：纯 AkShare 同步调用（东财 5-15s / 新浪 120s+），无并行，无代码缓存。

**改法**：加腾讯并行获取函数（跟 `stock.py` 的 `_from_tencent_threaded` 相同模式），并缓存港股代码列表。

```python
# hk.py 文件头部增加
import json, time, httpx
from concurrent.futures import ThreadPoolExecutor, as_completed

# 缓存变量
_hk_codes_cache = None
_hk_codes_ts = 0

def _load_hk_codes():
    global _hk_codes_cache, _hk_codes_ts
    now = time.time()
    if _hk_codes_cache and now - _hk_codes_ts < 86400:
        return _hk_codes_cache
    try:
        df = ak.stock_hk_spot_em()
        codes = [str(r['代码']) for _, r in df.iterrows()]
    except Exception:
        df = ak.stock_hk_spot()
        codes = [str(r.get('代码','')) for _, r in df.iterrows() if r.get('代码')]
    _hk_codes_cache = codes
    _hk_codes_ts = now
    return codes

def _from_tencent_hk(codes, workers=6):
    """腾讯港股并行获取 — 50只/请求, hk前缀"""
    from .utils import _safe_float
    batches = [codes[i:i+50] for i in range(0, len(codes), 50)]
    result = []
    def fetch_batch(batch):
        try:
            codes_clean = [str(c) for c in batch]
            url = 'https://qt.gtimg.cn/q=' + ','.join('hk' + c for c in codes_clean)
            resp = httpx.get(url, timeout=30)
            rows = []
            for line in resp.text.split('\n'):
                if '="' not in line: continue
                _, data = line.split('="', 1)
                fields = data.rstrip('";\n').split('~')
                if len(fields) < 35: continue
                rows.append({
                    '代码': fields[2], '名称': fields[1],
                    '最新价': _safe_float(fields[3]), '昨收': _safe_float(fields[4]),
                    '今开': _safe_float(fields[5]), '成交量': _safe_float(fields[6]),
                    '涨跌额': _safe_float(fields[31]) if len(fields)>31 else 0,
                    '涨跌幅': _safe_float(fields[32]) if len(fields)>32 else 0,
                    '最高': _safe_float(fields[33]) if len(fields)>33 else 0,
                    '最低': _safe_float(fields[34]) if len(fields)>34 else 0,
                    '成交额': _safe_float(fields[37]) if len(fields)>37 else 0,
                })
            return rows
        except Exception:
            return []
    with ThreadPoolExecutor(max_workers=workers) as ex:
        futures = [ex.submit(fetch_batch, b) for b in batches]
        for f in as_completed(futures): result.extend(f.result())
    return result
```

`fetch_shard` 和 `get_json` 优先用腾讯路径，失败回退 AkShare。

**验收**：港股模块首次加载 < 30s，后续 < 15s。

## #4 `kline.py` — httpx.Client 连接复用

**根因**：每次 HTTP 调用创建新连接，TLS 握手浪费 200-500ms。

**改法**：

```python
# kline.py 顶部增加模块级客户端
_tx_client = httpx.Client(timeout=30)

# _fetch_tencent 中 和 _fetch_tencent_minute 中，所有 httpx.get 改为：
resp = _tx_client.get(url)
```

**注意**：`httpx.Client` 不是线程安全的。如果多个 roller 线程同时调 `_fetch_tencent`，需要每个线程创建自己的 Client，或用锁保护。简单方案：在函数内用 `with httpx.Client(timeout=30) as client:` 而非模块级变量，避免线程问题但仍有连接池收益。

**验收**：K线 API 响应时间减少 ~200-500ms。

## #5 `main.py` — 启动并行预加载

**根因**：`_initial_load()` 串行加载 6 模块，最慢的美股卡住全部。

**改法**：

```python
# main.py _initial_load() 改为并行
def _initial_load():
    import concurrent.futures
    def _load_one(key, fn):
        try:
            raw = fn()
            data = json.loads(raw) if isinstance(raw, str) else raw
            n = SHARD_CFG[key]['n']
            if isinstance(data, list) and not data:
                print(f'[Preload] {key} empty, skipped')
                return
            if isinstance(data, dict) and not data:
                print(f'[Preload] {key} empty dict, skipped')
                return
            with _cache_lock:
                c = _cache.setdefault(key, {'shards': {}, 'cols': []})
                if isinstance(data, list):
                    chunk = max(1, len(data) // n)
                    for i in range(n):
                        shard_data = data[i*chunk:(i+1)*chunk] if i < n-1 else data[i*chunk:]
                        c['shards'][i] = {'data': shard_data, 'ts': time.time()}
                else:
                    c['shards'][0] = {'data': data, 'ts': time.time()}
            print(f'[Preload] {key} OK')
        except Exception as e:
            print(f'[Preload] {key}: {e}')

    modules = [
        ('news', get_news_json),
        ('index', get_index_json),
        ('etf', get_etf_json),       # 小模块优先
        ('hk', get_hk_json),
        ('stock', get_stock_json),
        ('us', get_us_json),         # 大模块最后
    ]
    with concurrent.futures.ThreadPoolExecutor(max_workers=6) as ex:
        futures = {ex.submit(_load_one, key, fn): key for key, fn in modules}
        concurrent.futures.wait(futures)
```

**验收**：服务启动后最快 10s 内可访问新闻/指数/ETF，最慢的美股独立加载不阻塞其他模块。

## #6 `main.py` — SSE 多客户端广播

**根因**：所有 SSE 连接共享一个 `Queue()`，消息被随机一个客户端消费，其余收不到。

**改法**：每个 SSE 连接创建自己的 `Queue`，roller 广播到所有连接。

```python
# main.py:_sse_queues 改为
_sse_queues = {m: [] for m in SHARD_CFG}  # list of client queues
_sse_lock = threading.Lock()

# _roller 中 把 put_nowait 改为广播：
with _sse_lock:
    for q in _sse_queues[key]:
        try:
            q.put_nowait({'shard': i, 'data': data, 'ts': time.time()})
        except queue.Full:
            pass

# stream_module 中 gen() 改为：
async def gen():
    import asyncio as _aio
    q = queue.Queue(maxsize=50)
    with _sse_lock:
        _sse_queues[module].append(q)
    try:
        while True:
            try:
                msg = await _aio.to_thread(q.get, True, 30)
                yield f'data: {json.dumps(msg)}\n\n'
            except queue.Empty:
                yield ': ping\n\n'
    finally:
        with _sse_lock:
            _sse_queues[module].remove(q)
```

**验收**：开两个浏览器标签页看同一模块，两个标签都收到实时数据更新。

## #7 `core.js` — flash 动画修复

**根因**：SSE 先给旧 DOM 加 flash class，再调用 `render()` 重建整个 tbody，flash 元素被销毁。

**改法**：render() 之后再给新 DOM 加 flash class。

```javascript
// core.js SSE onmessage 中，changed=true 分支
if (changed) {
    // ... (现有代码)
    if (tab === m) {
        rows = st.rows;
        updateTime = st.updateTime;
        render();  // 先重建 DOM
        // 再对新 DOM 加 flash
        shardRows.forEach(r => {
            let code = r['代码'] || r['交易对'];
            if (!code) return;
            Object.keys(r).forEach(k => {
                let nv = r[k];
                if (typeof nv === 'number' && (k.includes('涨跌') || k.includes('价'))) {
                    let cell = document.querySelector('td[data-code="' + code + '"][data-field="' + k + '"]');
                    if (cell) {
                        cell.classList.add('flash-up');
                        setTimeout(() => cell.classList.remove('flash-up'), 600);
                    }
                }
            });
        });
    }
}
```

**验收**：打开 A 股模块，交易时段可见价格单元格闪烁。

## #8 `indicators.py` — MA 数组长度修复

**根因**：`vals = [None] * (p - 1)` 初始固定长度，数据 < 周期时输出长度不等于输入。

**改法**：

```python
def calc_ma(closes, periods=(5, 10, 20, 60, 120, 250)):
    result = {}
    for p in periods:
        vals = []
        total = 0.0
        for i, v in enumerate(closes):
            total += v
            if i >= p:
                total -= closes[i - p]
            vals.append(round(total / p, 4) if i >= p - 1 else None)
        result[f'MA{p}'] = vals
    return result
```

**验收**：
```bash
python3 -c "from fetcher.indicators import calc_ma; r=calc_ma([100]*3); assert len(r['MA5'])==3 and len(r['MA250'])==3; print('OK')"
```

## #9 `utils.py` — fillna 字符串列

**根因**：`fillna(0)` 作用于所有非时间列，字符串列的 NaN → 整数 0。

**改法**：

```python
# utils.py _to_records 中
for col in df.columns:
    d = str(df[col].dtype)
    if any(k in d for k in ('datetime', 'timestamp', 'period', 'timedelta')):
        df[col] = df[col].astype(str)
    elif d == 'object':
        df[col] = df[col].fillna('')
    else:
        df[col] = df[col].fillna(0)
```

**验收**：
```bash
python3 -c "import pandas as pd; from fetcher.utils import _to_records; df=pd.DataFrame({'a':['x',None],'b':[1.0,None]}); r=_to_records(df); assert r[1]['a']==''; print('OK')"
```
